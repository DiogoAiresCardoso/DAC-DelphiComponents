procedure TDACDBGrid.InvalidateHoverCell(const ACell: TGridCoord);
var R: TRect;
begin
  if not HandleAllocated or (csDestroying in ComponentState) then Exit;
  R := DisplayCellRect(ACell.X, ACell.Y);
  if not R.IsEmpty then Winapi.Windows.InvalidateRect(Handle, @R, False);
end;

procedure TDACDBGrid.ClearHover;
var Old: TGridCoord;
begin
  Old := FHotCell;
  FHotCell := MakeGridCoord(-1, -1);
  FHotActionIndex := -1;
  if Cursor <> crDefault then Cursor := crDefault;
  if (Old.X >= 0) and HandleAllocated then
  begin
    InvalidateHoverCell(Old);
    if not (csDestroying in ComponentState) then Application.CancelHint;
  end;
end;

procedure TDACDBGrid.UpdateHover(const X, Y: Integer);
var Cell, Old: TGridCoord; Column: TDACDBGridColumn;
  View: TDACDBGridActionView; Context: TDACDBGridActionContext;
  Hot: Integer; NewCursor: TCursor; Hit: Boolean;
begin
  Cell := MakeGridCoord(-1, -1);
  Hot := -1; NewCursor := crDefault; Hit := False;
  if Enabled and not FLoading and HandleAllocated and
    (X >= 0) and (Y >= 0) and (X < ClientWidth) and (Y < ClientHeight) then
  begin
    Cell := MouseCoord(X, Y);
    if (Cell.Y >= FixedRows) and (not DataLink.Active or
      (Cell.Y - FixedRows >= DataLink.RecordCount)) then Cell := MakeGridCoord(-1, -1);
    if (Cell.Y = 0) and Sizing(X, Y) then
    begin Cell := MakeGridCoord(-1, -1); NewCursor := crHSplit; end
    else if (Cell.Y = 0) and (RawToDataColumn(Cell.X) >= 0) and
      (RawToDataColumn(Cell.X) < GetColumns.Count) and
      CanSortColumn(GetColumns[RawToDataColumn(Cell.X)]) and DataLink.Active then
      NewCursor := crHandPoint;
    Hit := ActionHit(X, Y, Column, View, Context);
    if Hit then
    begin
      Hot := View.ActionIndex;
      if View.Enabled then NewCursor := crHandPoint;
    end;
  end;
  if FActionMouse and (FPressedActionId <> '') and
    (not Hit or not View.Enabled or (Context.ActionId <> FPressedActionId) or
      (Context.ColumnKey <> FPressedActionColumn) or (Cell.Y <> FPressedActionRow)) then
    CancelActionInteraction;
  if not FHoverEnabled or FHeaderDragged or FHeaderSizing then
  begin Cell := MakeGridCoord(-1, -1); Hot := -1; end;
  if Cursor <> NewCursor then Cursor := NewCursor;
  if not SameGridCoord(Cell, FHotCell) or (Hot <> FHotActionIndex) then
  begin
    Old := FHotCell;
    FHotCell := Cell; FHotActionIndex := Hot;
    InvalidateHoverCell(Old);
    if not SameGridCoord(Old, Cell) then InvalidateHoverCell(Cell);
    Application.CancelHint;
  end;
end;

procedure TDACDBGrid.ClampColumnWidths;
var I, Minimum: Integer;
begin
  if FClampingColumns or (csLoading in ComponentState) or
    (csDestroying in ComponentState) then Exit;
  FClampingColumns := True;
  try
    for I := 0 to GetColumns.Count - 1 do
    begin
      Minimum := Pixels(GetColumns[I].MinWidth);
      if GetColumns[I].Width < Minimum then GetColumns[I].Width := Minimum;
    end;
  finally FClampingColumns := False; end;
end;

procedure TDACDBGrid.ColumnsChanged(Sender: TObject);
begin
  CancelActionInteraction;
  ClearHover;
  ClampColumnWidths;
  if (FLayoutManager <> nil) and not FLayoutManager.Applying then ValidateSort;
end;

function TDACDBGrid.CanSortColumn(AColumn: TDACDBGridColumn): Boolean;
begin
  Result := (AColumn <> nil) and AColumn.Sortable and
    (AColumn.FieldName <> '') and (AColumn.Field <> nil) and
    not AColumn.Field.IsBlob and (AColumn.Field.FieldKind = fkData);
end;

function TDACDBGrid.GetSortKey: string;
begin Result := FSortController.Key; end;

function TDACDBGrid.GetSortDirection: TDACDBGridSortDirection;
begin Result := FSortController.Direction; end;

procedure TDACDBGrid.ValidateSort;
var I: Integer;
begin
  if (FSortController = nil) or (FSortController.Direction = dgsNone) then Exit;
  for I := 0 to GetColumns.Count - 1 do
    if SameText(GetColumns[I].EffectiveLayoutKey, FSortController.Key) and
      CanSortColumn(GetColumns[I]) and
      SameText(GetColumns[I].FieldName, FSortController.FieldName) then Exit;
  FSortController.Clear;
end;

function TDACDBGrid.RequestSort(const AColumnIndex: Integer;
  const ADirection: TDACDBGridSortDirection): Boolean;
var Accepted: Boolean; Key, FieldName: string; I: Integer;
begin
  Result := False;
  if FSorting or not Enabled or FLoading or
    ([csLoading, csDesigning, csDestroying] * ComponentState <> []) or
    not DataLink.Active or (DataLink.DataSet = nil) or
    (DataLink.DataSet.State <> dsBrowse) or (AColumnIndex < 0) or
    (AColumnIndex >= GetColumns.Count) or not Assigned(FOnSortRequest) or
    not CanSortColumn(GetColumns[AColumnIndex]) then Exit;
  Key := GetColumns[AColumnIndex].EffectiveLayoutKey;
  FieldName := GetColumns[AColumnIndex].FieldName;
  CancelActionInteraction;
  ClearHover;
  Accepted := False;
  FSorting := True;
  try
    FOnSortRequest(Self, GetColumns[AColumnIndex], FieldName, ADirection, Accepted);
  finally FSorting := False; end;
  if not Accepted then Exit;
  // Resolve by identity again; a handler may have reorganized the columns.
  for I := 0 to GetColumns.Count - 1 do
    if SameText(Key, GetColumns[I].EffectiveLayoutKey) and
      SameText(FieldName, GetColumns[I].FieldName) and CanSortColumn(GetColumns[I]) then
    begin
      FSortController.Accept(Key, ADirection, FieldName);
      FPendingSortKey := '';
      if not FRestoringSort then FLayoutManager.ScheduleSave;
      Redraw;
      Exit(True);
    end;
  FSortController.Clear;
end;

procedure TDACDBGrid.RestoreLayoutSort(const AKey, AFieldName: string;
  const ADirection: TDACDBGridSortDirection);
begin
  FPendingSortKey := AKey; FPendingSortField := AFieldName;
  FPendingSortDirection := ADirection;
  TryPendingSort;
end;

procedure TDACDBGrid.TryPendingSort;
var I: Integer; Key, FieldName: string; Direction: TDACDBGridSortDirection;
begin
  if FSorting or FRestoringSort or (FPendingSortKey = '') or
    not DataLink.Active or (DataLink.DataSet = nil) then Exit;
  Key := FPendingSortKey; FieldName := FPendingSortField; Direction := FPendingSortDirection;
  FPendingSortKey := '';
  if Direction = dgsNone then Exit;
  for I := 0 to GetColumns.Count - 1 do
    if SameText(GetColumns[I].EffectiveLayoutKey, Key) and
      ((FieldName = '') or SameText(FieldName, GetColumns[I].FieldName)) then
    begin
      FRestoringSort := True;
      try RequestSort(I, Direction); finally FRestoringSort := False; end;
      Exit;
    end;
end;

procedure TDACDBGrid.Notification(AComponent: TComponent; Operation: TOperation);
begin
  if (Operation = opRemove) and (AComponent = FObservedDataSet) then
  begin
    FObservedDataSet := nil;
    if FSortController <> nil then FSortController.Clear;
    FPendingSortKey := '';
  end;
  inherited;
end;

procedure TDACDBGrid.TopLeftChanged;
begin
  CancelActionInteraction;
  ClearHover;
  inherited;
end;

procedure TDACDBGrid.CMHintShow(var AMessage: TCMHintShow);
var Column: TDACDBGridColumn; View: TDACDBGridActionView;
  Context: TDACDBGridActionContext; Cell: TGridCoord;
  Text: string; R: TRect; OldRecord, Index: Integer;
begin
  if not Enabled or FLoading or FHeaderDragged or FHeaderSizing then
  begin AMessage.Result := 1; Exit; end;
  if ActionHit(AMessage.HintInfo^.CursorPos.X, AMessage.HintInfo^.CursorPos.Y,
    Column, View, Context) then
  begin
    AMessage.HintInfo^.HintStr := Column.Actions[View.ActionIndex].Hint;
    AMessage.HintInfo^.CursorRect := View.Bounds;
    AMessage.Result := Ord(AMessage.HintInfo^.HintStr = '');
    Exit;
  end;
  Cell := MouseCoord(AMessage.HintInfo^.CursorPos.X, AMessage.HintInfo^.CursorPos.Y);
  Index := RawToDataColumn(Cell.X);
  if (Index < 0) or (Index >= GetColumns.Count) or (Cell.Y < 0) then
  begin inherited; Exit; end;
  Column := GetColumns[Index];
  R := DisplayCellRect(Cell.X, Cell.Y);
  Text := '';
  if Cell.Y = 0 then
  begin
    Text := Column.TitleHint;
    if Text = '' then Text := Column.Title.Caption;
  end
  else if DataLink.Active and (Cell.Y - FixedRows < DataLink.RecordCount) then
  begin
    Text := Column.Hint;
    if (Text = '') and (Column.Actions.Count = 0) then
    begin
      OldRecord := DataLink.ActiveRecord;
      try
        DataLink.ActiveRecord := Cell.Y - FixedRows;
        Text := TDACDBGridFieldTextResolver.Resolve(Column.Field);
      finally DataLink.ActiveRecord := OldRecord; end;
      if (DACMeasureDBGridText(Canvas.Handle, TDACComponentStyle.FontFamily,
        Text, DACDBGridDefaultFontSize, False, EffectivePPI).cx <= R.Width - Pixels(24)) and
        (Pos(#13, Text) = 0) and (Pos(#10, Text) = 0) then Text := '';
    end;
  end;
  AMessage.HintInfo^.HintStr := Text;
  AMessage.HintInfo^.CursorRect := R;
  AMessage.Result := Ord(Text = '');
end;
